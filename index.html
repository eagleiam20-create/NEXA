function findKnowledge(topic){
  const q=topic.toLowerCase();
  for(const [key,val] of Object.entries(knowledge))if(q.includes(key)||key.includes(q))return val;
  if(/photosynthesis|प्रकाश संश्लेषण/.test(q))return knowledge.photosynthesis;
  if(/gravity|gravitation|गुरुत्व/.test(q))return knowledge.gravity;
  if(/fraction|भिन्न/.test(q))return knowledge.fractions;
  if(/variable/.test(q))return knowledge.variables;
  return null;
}

/* Public Wikipedia reference search. */
async function getJSON(url){
  const controller=new AbortController(),timer=setTimeout(()=>controller.abort(),12000);
  try{const r=await fetch(url,{signal:controller.signal,headers:{Accept:"application/json"}});
    if(!r.ok)throw Error("HTTP "+r.status);return await r.json()
  }finally{clearTimeout(timer)}
}
async function searchWiki(query){
  const params=new URLSearchParams({action:"query",generator:"search",gsrsearch:query.slice(0,250),gsrnamespace:"0",gsrlimit:"5",prop:"extracts|info",exintro:"1",explaintext:"1",exchars:"1000",inprop:"url",format:"json",formatversion:"2",origin:"*"});
  const d=await getJSON("https://en.wikipedia.org/w/api.php?"+params);
  return ((d.query&&d.query.pages)||[]).sort((a,b)=>(a.index||999)-(b.index||999)).map(p=>({title:p.title,extract:p.extract||"No summary available.",url:p.fullurl||"https://en.wikipedia.org/wiki/"+encodeURIComponent(p.title.replace(/ /g,"_"))}));
}
$("#searchForm").addEventListener("submit",async e=>{

  loading(target,"Searching public references…");

No matching pages found. Try a shorter search.

    target.innerHTML="
Reference results: "+esc(q)+"
Public reference excerpts; verify important claims from original sources.
"+items.map((x,i)=>"

"+esc(x.extract)+"
"+link(x.url,"Read source")+"
").join("")+"
"+link("https://www.google.com/search?q="+encodeURIComponent(q),"Search the wider web")+"
";
  }catch(_){target.innerHTML="
Could not retrieve references. Check your connection or try again.
"+link("https://www.google.com/search?q="+encodeURIComponent(q),"Search the web instead")}
});
