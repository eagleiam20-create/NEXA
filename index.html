    email:["Subject: A clear summary of your purpose","Greeting","Opening: Explain why you are writing","Details: Give the necessary information","Action or question: State what you need","Polite closing and your name"],
    blog:["Title that tells readers what they will learn","Opening hook and reader problem","Context and key ideas","Practical examples or steps","Summary and useful next action"],
    presentation:["Slide 1: Title and objective","Slide 2: Why this topic matters","Slides 3–5: Main points and examples","Slide 6: Summary","Final slide: Questions or next steps"],
    project:["Project title and objective","Problem statement","Background and requirements","Method or implementation","Results, observations and limitations","Conclusion and references"],
    social:["Opening line or hook","One key message","Useful detail or example","Clear call to action if appropriate"]
  };
  const outline=`WRITING STARTER

TOPIC / PURPOSE
${topic}

TONE
${tone}

STRUCTURE
${structures[type].map((s,i)=>`${i+1}. ${s}`).join("\n")}

DRAFTING CHECKLIST
- Write for the intended audience.
- Use specific examples instead of unsupported claims.
- Check names, dates, figures and quotations.
- Keep the structure easy to follow.
- Review spelling and remove unnecessary repetition.
- Add references if you use outside information.

This is an outline, not a complete factual draft. Add your own details and verify any claims before publishing or submitting.`;
  showOutput("writingOutput",`
    ${escapeHTML(type.charAt(0).toUpperCase()+type.slice(1))} outline
    ${escapeHTML(outline)}
    ${copyButton(outline,"Copy outline")}`);
}

    ["Practise the core task","Repeat the main task and change one thing at a time."],
    ["Review problems","Test what works, note errors and fix the highest-priority issue."],
    ["Improve the result","Make the result clearer, more reliable and easier to use."],
    ["Show and get feedback","Share the result safely with a trusted person and note their feedback."],
    ["Learn the missing pieces","Focus on the biggest knowledge gap discovered so far."],
    ["Build the next feature","Add one small feature based on the goal, not just appearance."],

    ["Make a second version","Use feedback to make one meaningful improvement."],
    ["Organise the work","Keep files named clearly and record what changed."],
    ["Practise explaining it","Explain how it works, what it cannot do and what you learned."],
    ["Check the basics","Verify important functions and fix any obvious issues."],
    ["Document limitations","Write down what remains incomplete and what needs more research."],
    ["Review and choose the next goal","Summarise what is working, what you learned and your next step."]

  for(let i=0;iYour ${days}-day plan
    Goal: ${escapeHTML(goal)}
    Level: ${escapeHTML(level)} · Time: ${hours===30?"30 minutes":hours===60?"1 hour":"2 hours"} per day · Total: about ${totalMinutes} minutes
